const CURRENT_PROJECT = getProjectFromURL();
let HWTYPE_OPTIONS = [];
let HOSTS_MAP = {};
let CURRENT_HOSTNAME = null;

function rowHTML(h, group) {
  return `
    <tr>
      <td>${h.description || ""}</td>
      <td>${h.name || ""}</td>
      <td>${h.ip || ""}</td>
      <td>${h.mask_cidr || ""}</td>
      <td>
        <div class="action-buttons">
          <button onclick="openHostModal('${h.hostname}', '${group}')">Настроить</button>
          <button onclick="deleteHost('${h.hostname}')">Удалить</button>
        </div>
      </td>
    </tr>
  `;
}

function openHostModal(hostname, group) {
  const h = HOSTS_MAP[hostname];
  if (!h) return;

  CURRENT_HOSTNAME = hostname;

  document.getElementById("modalTitle").textContent = `Настройка хоста: ${hostname}`;
  document.getElementById("modal_ip").value = h.ip || "";
  document.getElementById("modal_description").value = h.description || "";
  document.getElementById("modal_mac").value = (h.mac || "").replace(/^'+|'+$/g, "");
  document.getElementById("modal_gate").value = h.gate || "";
  document.getElementById("modal_mask").value = h.mask || "";
  document.getElementById("modal_name").value = h.name || "";

  const groupSelect = document.getElementById("modal_group");
  if (groupSelect) groupSelect.value = group || "servers";

  const hw = document.getElementById("modal_hwtype");
  hw.innerHTML = HWTYPE_OPTIONS.map(x =>
    `<option value="${x}" ${x === h.hwtype ? "selected" : ""}>${x}</option>`
  ).join("");

  document.getElementById("hostModal").style.display = "block";
}

function closeHostModal() {
  document.getElementById("hostModal").style.display = "none";
  CURRENT_HOSTNAME = null;
}

function saveHost() {
  if (!CURRENT_HOSTNAME) return;

  const endpoint = CURRENT_HOSTNAME === "__NEW__" ? "/add_host" : "/update_host";

  fetch(endpoint, {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({
      project: CURRENT_PROJECT,
      hostname: CURRENT_HOSTNAME,
      values: {
        ip: document.getElementById("modal_ip").value,
        description: document.getElementById("modal_description").value,
        mac: document.getElementById("modal_mac").value,
        gate: document.getElementById("modal_gate").value,
        mask: document.getElementById("modal_mask").value,
        hwtype: document.getElementById("modal_hwtype").value,
        name: document.getElementById("modal_name").value,
        group: document.getElementById("modal_group").value
      }
    })
  })
  .then(r => r.json())
  .then(res => {
    if (res.ok) {
      closeHostModal();
      loadHostsInfo();
    } else {
      alert("Ошибка: " + (res.error || "неизвестно"));
    }
  });
}

function openAddHostModal() {
  CURRENT_HOSTNAME = "__NEW__";

  document.getElementById("modalTitle").textContent = "Добавить новый хост";
  document.getElementById("modal_ip").value = "";
  document.getElementById("modal_description").value = "";
  document.getElementById("modal_mac").value = "";
  document.getElementById("modal_gate").value = "";
  document.getElementById("modal_mask").value = "";
  document.getElementById("modal_name").value = "";
  document.getElementById("modal_group").value = "servers";

  const hw = document.getElementById("modal_hwtype");
  hw.innerHTML = HWTYPE_OPTIONS.map(x =>
    `<option value="${x}">${x}</option>`
  ).join("");

  document.getElementById("hostModal").style.display = "block";
}

function deleteHost(hostname) {
  if (!confirm(`Удалить хост "${hostname}"?`)) return;

  fetch("/delete_host", {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({
      project: CURRENT_PROJECT,
      hostname: hostname
    })
  })
  .then(r => r.json())
  .then(res => {
    if (res.ok) {
      loadHostsInfo();
    } else {
      alert("Ошибка удаления: " + (res.error || "неизвестно"));
    }
  });
}

function loadHostsInfo() {
  Promise.all([
    fetch("/data?project=" + encodeURIComponent(CURRENT_PROJECT)).then(r => r.json()),
    fetch("/hwtypes?project=" + encodeURIComponent(CURRENT_PROJECT)).then(r => r.json())
  ])
  .then(([data, hwtypes]) => {
    injectNav("hosts", data.projects || [], data.selected_project || "");
    HWTYPE_OPTIONS = hwtypes || [];

    const serversTable = document.getElementById("servers_table");
    const armsTable = document.getElementById("arms_table");

    HOSTS_MAP = {};

    serversTable.innerHTML = `
      <tr>
        <th>Название ФК</th>
        <th>Имя АПК</th>
        <th>IP</th>
        <th>Маска</th>
        <th>Действия</th>
      </tr>
    `;

    armsTable.innerHTML = `
      <tr>
        <th>Название ФК</th>
        <th>Имя АПК</th>
        <th>IP</th>
        <th>Маска</th>
        <th>Действия</th>
      </tr>
    `;

    data.hosts.servers.forEach(h => {
      HOSTS_MAP[h.hostname] = h;
      serversTable.innerHTML += rowHTML(h, "servers");
    });

    data.hosts.arm.forEach(h => {
      HOSTS_MAP[h.hostname] = h;
      armsTable.innerHTML += rowHTML(h, "arms");
    });
  });
}

loadHostsInfo();
